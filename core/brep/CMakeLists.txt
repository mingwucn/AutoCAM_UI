cmake_minimum_required(VERSION 3.20)
project(autocam_brep_core LANGUAGES CXX)
set(AUTOCAM_OCCT_BUILD "" CACHE PATH "Optional OCCT build tree (no installation needed)")
if(AUTOCAM_OCCT_BUILD)
  set(OpenCASCADE_INCLUDE_DIR "${AUTOCAM_OCCT_BUILD}/inc")
  if(EMSCRIPTEN)
    set(OpenCASCADE_INCLUDE_DIR "${AUTOCAM_OCCT_BUILD}/include/opencascade")
    set(OpenCASCADE_LIBRARY_DIR "${AUTOCAM_OCCT_BUILD}/lin32/clang/lib")
  elseif(WIN32)
    set(OpenCASCADE_LIBRARY_DIR "${AUTOCAM_OCCT_BUILD}/win64/gcc/lib")
  else()
    set(OpenCASCADE_LIBRARY_DIR "${AUTOCAM_OCCT_BUILD}/lin64/gcc/lib")
  endif()
else()
  find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms DataExchange)
endif()
add_library(autocam_brep_core STATIC material.cpp contours.cpp sweep_support.cpp analytic_pcurves.cpp regularize.cpp presentation.cpp envelope.cpp io.cpp session.cpp setup.cpp snapshot.cpp)
target_compile_features(autocam_brep_core PUBLIC cxx_std_17)
set_target_properties(autocam_brep_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(autocam_brep_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(autocam_brep_core PUBLIC ${OpenCASCADE_LIBRARY_DIR})
set(brep_libraries TKDESTEP TKXSBase TKDE TKHLR TKMesh TKBool TKBO TKShHealing TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR EMSCRIPTEN)
  target_link_libraries(autocam_brep_core PUBLIC "-Wl,--start-group" ${brep_libraries} "-Wl,--end-group")
else()
  target_link_libraries(autocam_brep_core PUBLIC ${brep_libraries})
endif()
if(WIN32)
  target_link_libraries(autocam_brep_core PUBLIC ws2_32 psapi)
endif()
if(EMSCRIPTEN)
  target_compile_options(autocam_brep_core PUBLIC -fexceptions)
  add_executable(autocam_brep c_api.cpp)
  set_target_properties(autocam_brep PROPERTIES SUFFIX ".mjs")
  target_link_options(autocam_brep PRIVATE --no-entry -fexceptions
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=8388608"
    "-sENVIRONMENT=web,worker,node" "-sEXPORTED_FUNCTIONS=['_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['cwrap','HEAPU8','HEAPU32','HEAPF32','HEAPF64']")
else()
  add_library(autocam_brep SHARED c_api.cpp)
  if(MINGW)
    target_link_options(autocam_brep PRIVATE -static-libgcc -static-libstdc++)
    get_filename_component(brep_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    find_file(brep_pthread_runtime libwinpthread-1.dll PATHS "${brep_compiler_bin}" REQUIRED)
    add_custom_command(TARGET autocam_brep POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${brep_pthread_runtime}" "$<TARGET_FILE_DIR:autocam_brep>/libwinpthread-1.dll")
  endif()
endif()
target_link_libraries(autocam_brep PRIVATE autocam_brep_core)
include(CTest)
