cmake_minimum_required(VERSION 3.20)
project(autocam_shadow_core LANGUAGES CXX)
if(EMSCRIPTEN)
  add_executable(autocam_shadow_core src/shadow_core.cpp)
else()
  add_library(autocam_shadow_core SHARED src/shadow_core.cpp)
endif()
target_include_directories(autocam_shadow_core PUBLIC include)
target_compile_features(autocam_shadow_core PUBLIC cxx_std_17)
target_compile_definitions(autocam_shadow_core PRIVATE SHADOW_CORE_BUILD)
set_target_properties(autocam_shadow_core PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MINGW)
  target_link_options(autocam_shadow_core PRIVATE -static-libgcc -static-libstdc++)
endif()
if(EMSCRIPTEN)
  set_target_properties(autocam_shadow_core PROPERTIES SUFFIX ".mjs")
  target_link_options(autocam_shadow_core PRIVATE
    "-sEXPORTED_FUNCTIONS=['_sg_api_version','_sg_status_message','_sg_milling_classify','_sg_turning_classify','_sg_apply','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['cwrap','HEAPU8','HEAPF64']" "-sMODULARIZE=1" "-sEXPORT_ES6=1")
endif()
